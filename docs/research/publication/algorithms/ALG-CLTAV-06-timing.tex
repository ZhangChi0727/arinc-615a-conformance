% ALG-CLTAV-06 — timed observation interpretation kernel (S6 kernel).
% Presentation of DD-029 / DD-033 / DD-036. Uses the approved T2/T5 rules; it
% invents no pairing policy and no error-estimation algorithm.
\begin{algorithm}[H]
\caption{\textsc{InterpretTimedObservation}: timed observation kernel (ALG-CLTAV-06, S6 kernel)}
\label{alg:cltav-06}
\SetKwInput{KwIn}{Require}
\SetKwInput{KwOut}{Ensure}
\KwIn{the single ownership result and the raw measurement interval from IF-OBS-INTERPRET; SelectSnapshot $\xi$; named obligation inputs $N_r$ (allowed interval) and $D_r$ (physical domain); declared match policy $\mathrm{Match}_r$; declared uncertainty $\varepsilon$}
\KwOut{a TimingResult: verdict in \{PASS, FAIL, INCONCLUSIVE, ERROR\}; the single pairing evidence; the interval used; validity}
\tcp{resolve $N_r$, $D_r$ and $\varepsilon$ from the named obligation and instance inputs, not from free symbols}
$\mathrm{pair}\leftarrow\mathrm{Match}_r(\mathrm{ownership},\xi)$\;\tcp{one pairing result feeds T2 and the no-response rule}
\lIf{$\mathrm{pair}$ is ambiguous}{\Return TimingResult$(\mathrm{verdict}=\mathrm{ERROR})$}
\uIf{$\mathrm{pair}$ is a legitimate trigger/response pair}{
  $M\leftarrow\mathrm{measurementInterval}\cap D_r$\;
  \lIf{$M=\varnothing$}{\Return TimingResult$(\mathrm{verdict}=\mathrm{ERROR})$}
  \uIf{$M\subseteq N_r$}{\Return TimingResult$(\mathrm{verdict}=\mathrm{PASS},\mathrm{evidence}=\mathrm{pair},M)$\;}
  \uElseIf{$M\cap N_r=\varnothing$}{\Return TimingResult$(\mathrm{verdict}=\mathrm{FAIL},\mathrm{evidence}=\mathrm{pair},M)$\;}
  \Return TimingResult$(\mathrm{verdict}=\mathrm{INCONCLUSIVE},\mathrm{evidence}=\mathrm{pair},M)$\;\tcp{open/closed endpoints preserved}
}
\tcp{a missing response forms a no-response only after a still-active instance reaches its deadline; this round not seeing it is not a timeout}
\uIf{the obligation is still active and expired}{
  \tcp{use the earliest admissible elapsed horizon (a scalar), with the requirement's boundary convention}
  \uIf{$N_r$ is closed at $U_r$}{$\mathrm{expired}\leftarrow\mathrm{lowerHorizon}>U_r$\;}
  \Else{$\mathrm{expired}\leftarrow\mathrm{lowerHorizon}\ge U_r$\;}
  \lIf{$\mathrm{expired}$}{\Return TimingResult$(\mathrm{verdict}=\mathrm{FAIL})$}
  \Return TimingResult$(\mathrm{verdict}=\mathrm{INCONCLUSIVE})$\;
}
\Return TimingResult$(\mathrm{verdict}=\mathrm{ERROR})$\;\tcp{a missing pair is not a point observation}
\end{algorithm}
