% ALG-CLTAV-03 — execution/observation coordination (S6 of ALG-CLTAV-01).
% Presentation of DD-029 / DD-033 / DD-036. Coordinates IF-OBS-INTERPRET, the
% timed kernel ALG-CLTAV-06 and IF-PREP-RECOVER; writes nothing.
\begin{algorithm}[H]
\caption{\textsc{InterpretOutcome}: execution and observation coordination (ALG-CLTAV-03, S6)}
\label{alg:cltav-03}
\SetKwInput{KwIn}{Require}
\SetKwInput{KwOut}{Ensure}
\KwIn{ExecutionResult $\textit{er}$; SelectSnapshot $\xi$; $\Gamma$; $\eta$; uncertainty reference $U$}
\KwOut{a total Outcome $z$: the execution effect, the interpretation effect class and the evaluated confirmation are combined by one decision table; skipped confirmation is NOT-EVALUATED, not an evaluated false}
\tcp{total constructor; the executed effect and the snapshot kind are copied, never left to an implicit global}
$z\leftarrow\textsc{NewOutcome}()$; $z.\mathrm{effect}\leftarrow\textit{er}.\mathrm{effect}$; $z.\mathrm{kind}\leftarrow\xi.\mathrm{actionKind}$; $z.\mathrm{actionId}\leftarrow\xi.\mathrm{actionId}$\;
\tcp{CONFIRMED-NOT-SENT and UNKNOWN-EFFECT do not call an observation or confirmation interface that did not run}
\uIf{$\textit{er}.\mathrm{effect}\in\{\texttt{CONFIRMED-NOT-SENT},\texttt{UNKNOWN-EFFECT}\}$}{\Return $z$\;}
$(I_z,\textit{effectClass},\textit{summaryConfirmed},\textit{postSummary},\mathrm{ownership},\mathrm{measurementInterval})\leftarrow$ \IFobs$(\Gamma,\textit{er}.\mathrm{record},\text{clocks},\varepsilon)$\;
\tcp{an invalid measurement or an interpretation ERROR is propagated to S7 and never confirmed}
\uIf{$\mathrm{ownership}$ is invalid}{$z.\mathrm{validity}\leftarrow\mathrm{ERROR}$; $z.\mathrm{effect}\leftarrow\texttt{UNKNOWN-EFFECT}$; \Return $z$\;}
$\mathrm{timing}\leftarrow\textsc{InterpretTimedObservation}(\mathrm{ownership},\mathrm{measurementInterval},\xi,N_r,D_r,\mathrm{Match}_r)$\;
\uIf{$\mathrm{timing}.\mathrm{verdict}=\mathrm{ERROR}$}{$z.\mathrm{validity}\leftarrow\mathrm{ERROR}$; $z.\mathrm{effect}\leftarrow\texttt{UNKNOWN-EFFECT}$; \Return $z$\;}
$z.\mathrm{validity}\leftarrow\mathrm{timing}.\mathrm{verdict}$; $z.I_z\leftarrow I_z$\;
\tcp{Prep/Recover confirmation runs BEFORE the effect is finalized, so a confirmed actual successor is not overwritten by an earlier UNKNOWN}
\If{$\xi.\mathrm{actionKind}\in\{\mathrm{PREP},\mathrm{RECOVER}\}$}{
  $z.\mathrm{prepResultEvaluated}\leftarrow\textbf{true}$; $(z.\mathrm{targetConfirmed},\textit{summaryConfirmed},z.\mathrm{prepErr},z.\mathrm{declaredTarget},z.\mathrm{evidence},\textit{postSummary})\leftarrow$ \IFprep$(\Gamma,\eta,\xi.\mathrm{actionId},\textit{er}.\mathrm{record})$\;
  \tcp{\IFprep returns values only; it does not write $\Gamma$ or $\eta$}
}
$z.\mathrm{summaryConfirmed}\leftarrow\textit{summaryConfirmed}$; $z.\mathrm{postSummary}\leftarrow\textit{postSummary}$\;
\tcp{one effect decision table: propagate the interpretation effect class; an unconfirmed successor or target becomes UNKNOWN-EFFECT}
\uIf{$\textit{effectClass}=\texttt{UNKNOWN-EFFECT}$ or $z.\mathrm{prepErr}$ or ($\xi.\mathrm{actionKind}$ is Prep and $z.\mathrm{prepResultEvaluated}$ and not $\textit{summaryConfirmed}$) or ($\xi.\mathrm{actionKind}$ is Recover and $z.\mathrm{prepResultEvaluated}$ and not $z.\mathrm{targetConfirmed}$) or ($\textit{effectClass}=\texttt{NONE}$ and not $\textit{summaryConfirmed}$)}{
  $z.\mathrm{effect}\leftarrow\texttt{UNKNOWN-EFFECT}$\;
}
\Else{$z.\mathrm{effect}\leftarrow\textit{effectClass}$; \tcp{a confirmed normal TEST/Prep/Recover keeps the interpretation effect class}\;}
\Return $z$; \tcp{values only; all $\Gamma$/$\eta$ writes stay in ALG-CLTAV-04 and S9}
\end{algorithm}
