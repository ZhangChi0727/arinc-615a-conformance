\begin{algorithm}[ht]
\caption{CL-TAV top-level session (ALG-CLTAV-01)}
\label{alg:cltav-01}
\SetKwInput{KwIn}{Require}
\SetKwInput{KwOut}{Ensure}
\KwIn{declared $H_0=\{h_{\mathrm{normal}}\}\cup H_{\mathrm{single}}$; SessionContext $\Gamma$ (unique state); HistoryHandle $\eta$; exactly one resource mode BUDGET$(c_{\min},B)$ xor ROUNDS$(K_{\max})$; action library; measurement uncertainty}
\KwOut{stop class, final $H$, $\eta$, session trace; singleton $h_{\mathrm{normal}}$ is not protocol PASS}
\tcp{\textbf{S0} initialize the session carriers; $q$ is $\Gamma.\mathrm{currentSummary}$ and retry is $\Gamma.\mathrm{retryCount}$; decision index $k$, billed charges and retry are distinct}
$(\Gamma,\eta,H)\leftarrow\textsc{Initialize}(H_0,\text{library},\text{uncertainty},\text{resourcePolicy})$\;
\While{true}{
  \tcp{\textbf{S1} entry stop gate before prediction; an already decided Admit A2--A5 or P1--P5 returns here}
  $\textit{stop}\leftarrow\textsc{EntryStop}(\Gamma,\eta,H)$\;
  \lIf{$\textit{stop}$ is decided}{\Return $\textsc{Finish}(\textit{stop},\Gamma,\eta,H)$}
  \tcp{\textbf{S2} predict before select; a PredictionGapError is TEST-scoped and does not by itself block eligible Recover or Prep}
  $\textit{prediction}\leftarrow\textsc{PredictCurrent}(\Gamma,\eta,H,\text{library},\text{uncertainty})$\;
  \lIf{$\textit{prediction}$ is PredictionGapError and Recover and Prep are ineligible}{\Return $\textsc{Finish}(\textsc{prediction-gap},\Gamma,\eta,H)$}
  \tcp{\textbf{S3} select the final non-ADMIT action, then freeze the immutable snapshot}
  $\textit{decision}\leftarrow\textsc{SelectAndAdmit}(\Gamma,\eta,H,\text{library},\textit{prediction})$\;
  \lIf{$\textit{decision}$ is an exit}{\Return $\textsc{Finish}(\textit{decision},\Gamma,\eta,H)$}
  \tcp{\textsc{FreezeFinalSelection}: freeze the final non-ADMIT action only after all selection rules}
  $\textit{qUsedAtSelect}\leftarrow\Gamma.\mathrm{currentSummary}$; $\textit{histVer}\leftarrow\eta.\mathrm{version}$; $\textit{classesUsedAtSelect}\leftarrow\mathcal{C}$; $\textit{actionId}\leftarrow t^\star$\;
  \tcp{\textbf{S4} charge once; retry cap $\neq$ budget exhaustion}
  $\Gamma\leftarrow\textsc{ChargeOnce}(\Gamma,\textit{actionId})$\;
  \tcp{\textbf{S5} execute the admitted action exactly once}
  $\textit{record}\leftarrow\textsc{ExecuteAndRecord}(\Gamma,\textit{actionId})$\;
  \tcp{\textbf{S6} interpret the execution record}
  $\textit{outcome}\leftarrow\textsc{InterpretOutcome}(\textit{record},\textit{actionId},\Gamma,\eta)$\;
  \tcp{\textbf{S7--S8} classify the effect and resolve retry or confirmed recovery}
  $(\Gamma,\eta,\textit{disposition})\leftarrow\textsc{ResolveOutcome}(\textit{outcome},\textit{actionId},\Gamma,\eta)$\;
  \lIf{$\textit{disposition}$ is STOP}{\Return $\textsc{Finish}(\textit{reason},\Gamma,\eta,H)$}
  \lIf{$\textit{disposition}$ is RETRY}{\textbf{continue}}
  \tcp{\textbf{S9} history reads the select snapshot; this is the sole commit of a confirmed successor summary}
  \If{$I_z$ is a valid observation class or $\textit{summaryConfirmed}$ is true}{
    $(\eta,H)\leftarrow$ \IFhist$(\eta,H,\textit{actionId},\textit{qUsedAtSelect},\textit{classesUsedAtSelect},I_z,\textit{postSummary})$\;
    \If{$\Gamma.\mathrm{qStatus}$ is KNOWN and $\textit{summaryConfirmed}$ is true and $\textit{postSummary}$ is confirmed}{
      $\Gamma.\mathrm{currentSummary}\leftarrow\textit{postSummary}$\;
    }
  }
  \tcp{\textbf{S10} post-update exclusive P-stops; if none, the next iteration starts at S1}
  $\textit{stop}\leftarrow\textsc{PostUpdateStop}(\Gamma,\eta,H)$\;
  \lIf{$\textit{stop}$ is decided}{\Return $\textsc{Finish}(\textit{stop},\Gamma,\eta,H)$}
}
\end{algorithm}
