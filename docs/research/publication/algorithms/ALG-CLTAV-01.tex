\begin{algorithm}[H]
\caption{CL-TAV top-level session (ALG-CLTAV-01)}
\label{alg:cltav-01}
\SetKwInput{KwIn}{Require}
\SetKwInput{KwOut}{Ensure}
\KwIn{$H_0=\{h_{\mathrm{normal}}\}\cup H_{\mathrm{single}}$; SessionContext $\Gamma$ (unique state); HistoryHandle $\eta$; exactly one resource mode BUDGET$(c_{\min},B)$ xor ROUNDS$(K_{\max})$; action library $L$; uncertainty reference $U$}
\KwOut{stop class, final $H$, $\eta$, session trace; singleton $h_{\mathrm{normal}}$ is not protocol PASS}
\tcp{\textbf{S0} \textsc{Initialize}: state, candidates, history, resource, retry, uncertainty reference}
$(\Gamma,\eta,H)\leftarrow\textsc{Initialize}(H_0,L,U,\text{resourcePolicy})$\;
\While{true}{
  \tcp{\textbf{S1} \textsc{EntryStop}: maps to the existing A2--A5 / P1--P5 decision table (Appendix C)}
  $\textit{stop}\leftarrow$ \IFres$(\Gamma,H,\eta)$ with \IFeq$(\Gamma,H,\eta)$\;
  \lIf{$\textit{stop}\neq\texttt{undecided}$}{\Return $\textsc{Finish}(\textit{stop},\Gamma,\eta,H)$}
  \tcp{\textbf{S2} \textsc{PredictCurrent}: typed prediction; the gap tag is handled before any projection}
  $\textit{pred}\leftarrow\textsc{PredictCurrent}(\Gamma,\eta,H,U,L)$\;
  \tcp{\textbf{S3} \textsc{SelectAndAdmit}: GAP and OK share this admission; tagged exit or spec error returns with its reason}
  $\textit{dec}\leftarrow\textsc{SelectAndAdmit}(\Gamma,\eta,H,L,\textit{pred})$\;
  \lIf{$\textit{dec}.\mathrm{kind}\in\{\texttt{EXIT},\texttt{SPEC-ERROR}\}$}{\Return $\textsc{Finish}(\textit{dec}.\mathrm{reason},\Gamma,\eta,H)$}
  \tcp{\textbf{S3-SNAP} \textsc{SelectSnapshot} freezes $\xi$ (actionId, actionKind, qUsedAtSelect, historyVersion, classesUsedAtSelect, uncertaintyRef); the action kind is $\textit{dec}.\mathrm{actionKind}$, never the exit/action discriminant}
  $\xi\leftarrow\textsc{SelectSnapshot}(\textit{dec}.\mathrm{actionId},\ \textit{dec}.\mathrm{actionKind},\ \Gamma.\mathrm{currentSummary},\ \eta.\mathrm{version},\ \textit{dec}.\mathrm{classesUsed},\ U)$\;
  \tcp{\textbf{S4} \textsc{ChargeOnce}: one charge under the selected mode; retry cap $\neq$ budget exhaustion}
  $\Gamma\leftarrow\textsc{ChargeOnce}(\Gamma,\xi)$\;
  \tcp{\textbf{S5} \textsc{ExecuteAndRecord}: execute the snapshot action exactly once}
  $\textit{er}\leftarrow$ \IFexec$(\Gamma,\xi)$\;
  \tcp{\textbf{S6} \textsc{InterpretOutcome}: ALG-CLTAV-03 coordinates; ALG-CLTAV-06 interprets timing}
  $z\leftarrow\textsc{InterpretOutcome}(\textit{er},\xi,\Gamma,\eta,U)$\;
  \tcp{\textbf{S7--S8} \textsc{ResolveOutcome}: a total Resolution; every variant consumes the same returned contexts}
  $\delta\leftarrow\textsc{ResolveOutcome}(z,\xi,\Gamma,\eta)$\;
  \tcp{adopt the returned context and the normalized outcome before branching; a RETRY must keep the incremented retry and UNKNOWN state}
  $\Gamma\leftarrow\delta.\Gamma$; $\eta\leftarrow\delta.\eta$; $z\leftarrow\delta.\mathrm{outcome}$\;
  \lIf{$\delta.\mathrm{control}=\texttt{STOP}$}{\Return $\textsc{Finish}(\delta.\mathrm{reason},\Gamma,\eta,H)$}
  \lIf{$\delta.\mathrm{control}=\texttt{RETRY}$}{\textbf{continue} to S1 with the adopted context}
  \tcp{\textbf{S9} the S9 guard is an executable condition: a valid observation or a confirmed successor summary}
  \If{$z.I_z$ is a valid observation class or $z.\mathrm{summaryConfirmed}$ is true}{
    $(\textit{status},\eta,H,\Gamma)\leftarrow\textsc{CommitCompatibleUpdate}(\xi,z,\Gamma,\eta,H)$\;
    \lIf{$\textit{status}=\texttt{SPEC-ERROR}$}{\Return $\textsc{Finish}(\text{history version or intersection mismatch},\Gamma,\eta,H)$\;}
  }
  \tcp{\textbf{S10} \textsc{PostUpdateStop}: existing post-update stop priority; otherwise the next iteration is S1}
  $\textit{stop}\leftarrow$ \IFres$(\Gamma,H,\eta)$ with \IFeq$(\Gamma,H,\eta)$\;
  \lIf{$\textit{stop}\neq\texttt{undecided}$}{\Return $\textsc{Finish}(\textit{stop},\Gamma,\eta,H)$}
}
\end{algorithm}
