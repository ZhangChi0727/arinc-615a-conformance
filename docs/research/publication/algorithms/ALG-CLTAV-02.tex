% backend is neither selected nor implemented; its return tag is handled first.
\begin{algorithm}[H]
\caption{\textsc{PredictCurrent}: current-observation prediction wrapper (ALG-CLTAV-02, S2)}
\label{alg:cltav-02}
\SetKwInput{KwIn}{Require}
\SetKwInput{KwOut}{Ensure}
\KwIn{SessionContext $\Gamma$; HistoryHandle $\eta$; current $H$; uncertainty reference $U$; action library $L$}
\KwOut{PredictionResult: status OK with classesByTest and metadata, or status GAP with a reason; the session's own $\eta$ is read, not rebuilt}
$\textit{raw}\leftarrow$ \IFpred$(\Gamma,\eta,H,U,L)$\;
\tcp{branch on the backend result tag BEFORE projection; an error sentinel is not a set of observation classes}
\uIf{$\textit{raw}.\mathrm{tag}=\texttt{GAP}$}{
  \Return PredictionResult$(\mathrm{status}=\texttt{GAP},\mathrm{reason}=\textit{raw}.\mathrm{reason})$; \tcp{the gap stays TEST-scoped and does not forbid eligible Prep/Recover}
}
$\textit{classes}\leftarrow\textsc{ProjectOntoCurrentH}(\textit{raw}.\mathrm{classes},H)$; \tcp{an empty projection from an excluded hypothesis is not a surviving $score=0$ class}
\Return PredictionResult$(\mathrm{status}=\texttt{OK},\mathrm{classesByTest}=\textit{classes},\mathrm{historyVersionUsed}=\eta.\mathrm{version},\mathrm{uncertaintyRef}=U)$\;
\end{algorithm}
