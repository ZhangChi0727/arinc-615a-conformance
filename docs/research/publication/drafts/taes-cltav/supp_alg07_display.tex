\begin{algorithm}[H]
\caption{\textsc{CommitCompatibleUpdate}: compatible update and summary commit (S6 / ALG-CLTAV-07)}
\label{alg:cltav-07}
\KwIn{snapshot $\xi$; Outcome $z$; contexts $\Gamma,\eta,H$}
\KwOut{UpdateResult $(status,\eta',H',\Gamma')$}
$\Gamma'\leftarrow\Gamma$\;
\lIf{$\xi.\mathrm{historyVersion}\neq\eta.\mathrm{version}$}{\Return SPEC-ERROR with unchanged $\eta,H,\Gamma'$}
\eIf{$z.I_z$ is a valid compatible class}{
  $(\eta_c,H_c)\leftarrow$ \IFhist{} with:\;
  \Indp $\eta,H,\xi.\mathrm{actionId},\xi.\mathrm{qUsedAtSelect},\xi.\mathrm{classesUsedAtSelect},$\;
  $\xi.\mathrm{historyVersion},z.I_z,z.\mathrm{postSummary}$\; \Indm
  \lIf{$H_c\neq H\cap z.I_z$}{\Return SPEC-ERROR (intersection mismatch)}
  $\eta'\leftarrow\eta_c$; $H'\leftarrow H_c$\;
}{
  $(\eta',H')\leftarrow$ \IFhist{} with the same snapshot fields, NONE, and $z.\mathrm{postSummary}$\;
  $H'\leftarrow H$\;
}
\lIf{$\Gamma'.\mathrm{qStatus}$ is KNOWN and $z.\mathrm{summaryConfirmed}$}{\(\Gamma'.\mathrm{currentSummary}\leftarrow z.\mathrm{postSummary}\)}
\Return OK with $(\eta',H',\Gamma')$\;
