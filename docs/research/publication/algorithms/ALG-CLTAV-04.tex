\begin{algorithm}[ht]
\caption{\textsc{ResolveOutcome}: outcome resolution and confirmed recovery (ALG-CLTAV-04, S7--S8)}
\label{alg:cltav-04}
\SetKwInput{KwIn}{Require}
\SetKwInput{KwOut}{Ensure}
\KwIn{structured outcome $z$; immutable select snapshot $\xi$; SessionContext $\Gamma$; HistoryHandle $\eta$}
\KwOut{control class COMMIT / RETRY / STOP; updated $\Gamma,\eta$}
\uIf{$\textit{effect}\in\{\texttt{CONFIRMED-NOT-SENT},\texttt{UNKNOWN-EFFECT}\}$ or $\textit{prepErr}$ or ($\textit{kind}$ is Prep and $\textit{prepResultEvaluated}$ and $\textit{summaryConfirmed}$ is false) or ($\textit{actionId}$ is Recover and $\textit{prepResultEvaluated}$ and $\textit{targetConfirmed}$ is false)}{
  \tcp{\textbf{S7} classify the effect; do not fold \texttt{CONFIRMED-NOT-SENT} into UNKNOWN}
  do not exclude $H$\;
  \uIf{$\textit{prepErr}$ or $\textit{effect}=\texttt{UNKNOWN-EFFECT}$ or ($\textit{kind}$ is Prep and $\textit{prepResultEvaluated}$ and $\textit{summaryConfirmed}$ is false)}{
    mark $\eta$ conservative-unknown; $\Gamma.\mathrm{qStatus}\leftarrow$ UNKNOWN; clear $\Gamma.\mathrm{currentSummary}$\;
  }
  \uElseIf{$\textit{effect}=\texttt{CONFIRMED-NOT-SENT}$}{
    leave $\Gamma.\mathrm{qStatus}$ and compatible history unchanged\;
  }
  \Else{
    \tcp{unconfirmed Recover target}
    $\Gamma.\mathrm{qStatus}$ stays UNKNOWN; clear $\Gamma.\mathrm{currentSummary}$; mark $\eta$ conservative-unknown\;
  }
  $\Gamma.\mathrm{retryCount}\leftarrow\Gamma.\mathrm{retryCount}+1$\;
  \lIf{$\Gamma.\mathrm{retryCount}\ge\Gamma.\mathrm{retryCap}$}{\Return STOP(Stop-Error); this is decided before A4/A5 and is not Stop-Budget}
  \Return RETRY\;
}
\uElseIf{$\textit{actionId}$ is Recover}{
  \tcp{\textbf{S8} sole writer of KNOWN$(\textit{declaredTarget},\textit{evidence})$}
  $\Gamma.\mathrm{qStatus}\leftarrow$ KNOWN$(\textit{declaredTarget},\textit{evidence})$\;
  $\textit{postSummary}\leftarrow(\textit{declaredTarget},\textit{evidence})$\;
}
\Return COMMIT; no path here starts prediction directly, S9 remains the successor-summary commit\;
\end{algorithm}
