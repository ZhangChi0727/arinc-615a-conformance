\subsection{Definitions and assumptions}

\begin{definition}[Conservative outer set]
\label{def:cons-obs}
For hypothesis \(h\), action \(t\) and summary \(q\), \(O(h,t,q)\) is a
conservative outer set of feasible observations. \(Z_k\) is the measured
feasible-observation set after timing, ownership and pairing checks.
\end{definition}

\begin{definition}[Compatible class]
\label{def:compat}
Prediction and interpretation share one uncertainty reference \(U\). The
interpretation/whole-history bridge induces the compatible hypothesis class
\(C_k\) (accepted field \(z.I_z\)); \(C_k\) is not a measurement set. A
currently valid class is the nonempty projection \(H\cap C_k\).
\end{definition}

\begin{assumption}[Exclusive resource mode]
\label{ass:budget}
A session is in exactly one mode, \(\mathrm{BUDGET}(c_{\min},B)\) or
\(\mathrm{ROUNDS}(K_{\max})\), never both:
\begin{equation}
\label{eq:budget}
\bigl(\mathrm{BUDGET}(c_{\min},B)\ \mathrm{xor}\ \mathrm{ROUNDS}(K_{\max})\bigr)=1.
\end{equation}
Every charged TEST, Prep, Recover and \texttt{ERROR} spends that mode. Retry
caps are not a second budget.
\end{assumption}

\subsection{Update and selection}

Compatible update on a valid class is
\begin{equation}
\label{eq:update}
H_{k+1}=H_k\cap C_k,\qquad H_{k+1}\subseteq H_k.
\end{equation}
\texttt{ERROR} and identity Prep/Recover leave \(H\) unchanged. The one-step
score on currently valid nonempty classes of an affordable distinguishing
TEST \(t\) is
\begin{equation}
\label{eq:score}
s(t)=\max_{C\in\mathcal{C}_t}\lvert C\rvert.
\end{equation}
Empty \(S\) is never \(\arg\min\). On nonempty finite \(T_{\mathrm{cur}}\),
\(t^{\star}\) is a least element of \((s(t),\mathrm{cost}(t),\mathrm{id})\).
Timing uses T5~\cite{jcgm106}: \(I_{\mathrm{obs}}\subseteq I_r\) is PASS;
\(I_{\mathrm{obs}}\cap I_r=\varnothing\) is FAIL; otherwise INCONCLUSIVE. An
ambiguous pair is \texttt{ERROR}, not IUT FAIL.

This contract is intentionally weaker than a universal uncertain-observation
diagnosis theorem~\cite{koehl2023}. CL-TAV claims only the consistency and
retention property below under R1--R5.

\subsection{Propositions}

\begin{proposition}[Monotonicity]
\label{prop:mono}
Equation~\eqref{eq:update} gives \(H_{k+1}\subseteq H_k\). Without a valid class
there is no add or delete. An in-domain singleton is not protocol PASS.
\end{proposition}
\begin{IEEEproof}
On a valid class, \(H_{k+1}=H_k\cap C_k\subseteq H_k\). \texttt{ERROR} and
identity Prep/Recover leave \(H\) unchanged, so \(H_{k+1}=H_k\subseteq H_k\).
\end{IEEEproof}

\begin{proposition}[True-fault retention]
\label{prop:retain}
Assume (R1) \(h^{\star}\in H_0\); (R2) one state path is consistent with the
executed history; (R3) \(O(h^{\star},t_k,q_k)\) is a conservative outer set;
(R4) the true observation belongs to both \(O(h^{\star},t_k,q_k)\) and \(Z_k\);
and (R5) ownership and pairing are valid. Then \(h^{\star}\in H_{k+1}\).
\end{proposition}
\begin{IEEEproof}
The declared interpretation and whole-history compatibility contract therefore
places \(h^{\star}\) in \(C_k\). Thus \(h^{\star}\in H_k\cap C_k\), and induction
retains it in \(H_{k+1}\). Out-of-domain faults may still mimic an in-domain
survivor.
\end{IEEEproof}

\begin{proposition}[One-step minimax]
\label{prop:c2}
On nonempty finite \(T_{\mathrm{cur}}\), \(t^{\star}\) is a least element of
\((s(t),\mathrm{cost}(t),\mathrm{id})\). Empty \(S\) is never \(\arg\min\).
This is not a global policy, does not prove future distinguishability, and
does not guarantee shrinkage for every observation class.
\end{proposition}
\begin{IEEEproof}
\(\mathcal{C}_t\) is a finite family of currently valid nonempty classes of an
affordable distinguishing TEST, so \(s(t)\) in~\eqref{eq:score} is a well-defined
nonnegative integer. The lexicographic order on
\((s(t),\mathrm{cost}(t),\mathrm{id})\) is a total order on finite nonempty
\(T_{\mathrm{cur}}\), hence a least element exists. If \(S=\emptyset\), the
session classifies Admit A2--A5 rather than minimizing over an empty set.
\end{IEEEproof}

\begin{proposition}[Termination]
\label{prop:term}
Under Assumption~\ref{ass:budget} every charged TEST, Prep, Recover and
\texttt{ERROR} spends the selected mode, hence at most
\(\lfloor B/c_{\min}\rfloor\) or \(K_{\max}\) billed executions.
\end{proposition}
\begin{IEEEproof}
Each billed execution consumes a strictly positive quantum or one round, so
the remaining resource is a nonnegative integer that strictly decreases.
Named stops in Table~\ref{tab:stops} are the only exits. Wall-clock waits still
need their own timeouts; those timeouts are not claimed implemented.
\end{IEEEproof}

The update disposition is evaluated before the mutable stop label: a rejected
or \texttt{ERROR} update preserves \(H\), while a valid empty or singleton
update forces Stop-Empty or Stop-Singleton. \texttt{CONFIRMED-NOT-SENT} does
not assign UNKNOWN. \texttt{UNKNOWN-EFFECT} must not be read back as the
previous KNOWN summary.
