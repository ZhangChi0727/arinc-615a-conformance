\begin{algorithm}[ht]
\caption{\textsc{InterpretOutcome}: execution and observation interpretation (ALG-CLTAV-03, S6)}
\label{alg:cltav-03}
\SetKwInput{KwIn}{Require}
\SetKwInput{KwOut}{Ensure}
\KwIn{execution result; immutable select snapshot $\xi$ (see the symbol table); session and history context; declared measurement uncertainty}
\KwOut{structured outcome: execution effect class, observation validity and feasible classes, whether the successor summary is confirmed, the confirmed successor $\textit{postSummary}$, whether a Prep/Recover confirmation actually ran, $\textit{targetConfirmed}$, $\textit{prepErr}$, and evidence references}
\tcp{CONFIRMED-NOT-SENT and UNKNOWN-EFFECT do not call an observation or confirmation interface that was not executed; an unset confirmation field is not an evaluated failure}
\eIf{$\textit{effect}\in\{\texttt{CONFIRMED-NOT-SENT},\texttt{UNKNOWN-EFFECT}\}$}{
  skip interpretation; keep the successor summary unconfirmed\;
}{
  $(I_z,\textit{effect},\textit{summaryConfirmed},\textit{postSummary})\leftarrow$ \IFobs$(\Gamma,\textit{record})$\;
}
\tcp{a normal TEST explicitly receives summaryConfirmed and postSummary; event ownership, pairing and interval treatment follow the method contract}
\If{$\textit{kind}\in\{\mathrm{PREP},\mathrm{RECOVER}\}$ and $\textit{effect}\notin\{\texttt{CONFIRMED-NOT-SENT},\texttt{UNKNOWN-EFFECT}\}$}{
  $(\textit{targetConfirmed},\textit{summaryConfirmed},\textit{prepErr},\textit{declaredTarget},\textit{evidence},\textit{postSummary})\leftarrow$ \IFprep$(\Gamma,\eta,\textit{actionId},\textit{record})$; $\textit{prepResultEvaluated}\leftarrow$ true\;
  \tcp{\IFprep returns values; it does not write $\Gamma$ or $\eta$. A Prep whose target is not reached may still have a confirmed actual successor}
  \lIf{$\textit{kind}$ is Prep and $\textit{prepResultEvaluated}$ and $\textit{summaryConfirmed}$ is false}{record an unconfirmed Prep successor summary for S7\;}
}
\Return the structured outcome, including whether the unconfirmed Prep successor condition ($\textit{kind}$ is Prep and $\textit{prepResultEvaluated}$ and $\textit{summaryConfirmed}$ is false) holds\;
\end{algorithm}
